- [ ] **Step 3: Write 0039 migration**

Create `supabase/migrations/0039_logistics_confirm_v3_auto_skip.sql`. CREATE OR REPLACE `logistics_confirm_proceed_request_v3`. Body includes:

```sql
-- Existing v3 confirm logic from 0034 (resolve order, create threads at
-- 'awaiting_logistics_action', etc.) — preserved verbatim.

-- NEW BLOCK 1: Compute per-thread stock-sufficiency
WITH thread_demand AS (
  SELECT t.id AS thread_id, ol.sku_id, sum(ol.qty) AS need
    FROM order_supplier_threads t
    JOIN order_lines ol ON ol.order_id = t.order_id
   WHERE t.order_id = p_order_id
     AND ol.supplier_id = t.supplier_id  -- per-thread filter
   GROUP BY t.id, ol.sku_id
),
stock AS (
  SELECT sku_id, sum(available_qty) AS avail
    FROM stock_balances
   WHERE sku_id IN (SELECT sku_id FROM thread_demand)
   GROUP BY sku_id
)
SELECT bool_and(td.need <= coalesce(s.avail, 0))
  INTO v_all_sufficient
  FROM thread_demand td
  LEFT JOIN stock s ON s.sku_id = td.sku_id;

-- NEW BLOCK 2: If all sufficient, atomically reserve + promote
IF v_all_sufficient THEN
  -- FOR UPDATE on stock_balances rows to prevent concurrent over-reserve
  PERFORM 1
    FROM stock_balances
   WHERE sku_id IN (SELECT sku_id FROM thread_demand)
   FOR UPDATE;

  -- Re-check after lock (defense against TOCTOU)
  SELECT bool_and(td.need <= coalesce(s.avail, 0))
    INTO v_all_sufficient
    FROM thread_demand td
    LEFT JOIN (
      SELECT sku_id, sum(available_qty) AS avail
        FROM stock_balances
       WHERE sku_id IN (SELECT sku_id FROM thread_demand)
       GROUP BY sku_id
    ) s ON s.sku_id = td.sku_id;

  IF v_all_sufficient THEN
    -- Decrement stock + write movements
    UPDATE stock_balances sb
       SET available_qty = sb.available_qty - td.need
      FROM (
        SELECT sku_id, sum(need) AS need FROM thread_demand GROUP BY sku_id
      ) td
     WHERE sb.sku_id = td.sku_id;

    INSERT INTO stock_movements (kind, reason, sku_id, qty, order_id, created_by)
    SELECT 'out', 'reserve_from_buffer', td.sku_id, td.need, p_order_id, auth.uid()
      FROM (
        SELECT sku_id, sum(need) AS need FROM thread_demand GROUP BY sku_id
      ) td;

    -- Promote all threads to ready_to_dispatch
    UPDATE order_supplier_threads
       SET logistics_stage = 'ready_to_dispatch'
     WHERE order_id = p_order_id;

    -- orders_rollup_stage() trigger (0036) fires → orders.logistics_stage = 'ready_to_dispatch'
  END IF;
END IF;

-- Return existing v3 response shape, with auto_skipped boolean
RETURN jsonb_build_object(
  'order_id', p_order_id,
  'threads', (SELECT jsonb_agg(row_to_json(t)) FROM order_supplier_threads t WHERE t.order_id = p_order_id),
  'auto_skipped', v_all_sufficient,
  'po_id', null  -- no ghost PO per Phase 4.5a design decision
);
```

Adapt to actual function signature in 0034. Use 0034 file header style for the new migration. Document the "no ghost PO" decision in header comments.
